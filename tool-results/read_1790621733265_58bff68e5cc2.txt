    90→const BIRTH_MONTH_KEY = "planner.fal.birth-month";
    91→const BIRTH_MONTH_EVENT = "planner:birth-month-changed";
    92→
    93→let birthMonthCache: number | null | undefined;
    94→
    95→function readBirthMonth(): number | null {
    96→  if (typeof window === "undefined") return null;
    97→  if (birthMonthCache === undefined) {
    98→    const raw = window.localStorage.getItem(BIRTH_MONTH_KEY);
    99→    const n = raw === null ? NaN : parseInt(raw, 10);
   100→    birthMonthCache = Number.isInteger(n) && n >= 1 && n <= 12 ? n : null;
   101→  }
   102→  return birthMonthCache;
   103→}
   104→
   105→function subscribeBirthMonth(cb: () => void) {
   106→  const handler = () => {
   107→    birthMonthCache = undefined; // کش را باطل کن (به‌ویژه برای رویداد storage بین تب‌ها)
   108→    cb();
   109→  };
   110→  window.addEventListener(BIRTH_MONTH_EVENT, handler);
   111→  window.addEventListener("storage", handler);
   112→  return () => {
   113→    window.removeEventListener(BIRTH_MONTH_EVENT, handler);
   114→    window.removeEventListener("storage", handler);
   115→  };
   116→}
   117→
   118→function writeBirthMonth(m: number | null) {
   119→  if (typeof window === "undefined") return;
   120→  if (m === null) window.localStorage.removeItem(BIRTH_MONTH_KEY);
   121→  else window.localStorage.setItem(BIRTH_MONTH_KEY, String(m));
   122→  birthMonthCache = m;
   123→  window.dispatchEvent(new CustomEvent(BIRTH_MONTH_EVENT));
   124→}

   461→
   462→function DateConverter() {
   463→  const mounted = useMounted();
   464→  const [src, setSrc] = useState<CalKey>("jalali");
   465→  // فیلدهای ورودی — تا وقتی دست‌نخورده باشد، «امروز» نمایش داده می‌شود
   466→  const [fields, setFields] = useState<{
   467→    y: string;
   468→    m: number;
   469→    d: string;
   470→    touched: boolean;
   471→  }>({ y: "", m: 0, d: "", touched: false });
   472→
   473→  // مقدار «امروز» در تقویم مبدا — فقط بعد از mount
   474→  const def = useMemo(() => {
   475→    if (!mounted) return { y: "", m: 0, d: "" };
   476→    const now = new Date();
   477→    if (src === "jalali") {
   478→      const j = toJalali(now);
   479→      return { y: String(j.jy), m: j.jm, d: String(j.jd) };
   480→    }
   481→    if (src === "gregorian") {
   482→      return {
   483→        y: String(now.getFullYear()),
   484→        m: now.getMonth() + 1,
   485→        d: String(now.getDate()),
   486→      };
   487→    }
   488→    const h = toHijri(now);
   489→    return { y: String(h.hy), m: h.hm, d: String(h.hd) };
   490→  }, [mounted, src]);
   491→
   492→  const shown = fields.touched
   493→    ? fields
   494→    : { y: fields.y || def.y, m: fields.m || def.m, d: fields.d || def.d };
   495→
   496→  function update(patch: Partial<{ y: string; m: number; d: string }>) {
   497→    setFields((prev) => {
   498→      // اولین ویرایش از مقادیر نمایش‌داده‌شده (امروز) شروع می‌شود تا ماه/روز پاک نشوند
   499→      const base = prev.touched
   500→        ? prev
   501→        : { y: shown.y, m: shown.m, d: shown.d, touched: false };
   502→      return { ...base, ...patch, touched: true };
   503→    });
   504→  }
   505→
   506→  function resetToToday() {
   507→    setFields((prev) => ({ ...prev, touched: false }));
   508→  }
   509→
   510→  function switchSource(next: CalKey) {
   511→    setSrc(next);
   512→    setFields((prev) => ({ ...prev, touched: false }));
   513→  }
   514→
   515→  // اعتبارسنجی و تاریخ مبدا
   516→  const result = useMemo(() => {
   517→    const yN = parseInt(faToEn(shown.y), 10);
   518→    const dN = parseInt(faToEn(shown.d), 10);
   519→    if (!shown.y || !shown.d || !shown.m || Number.isNaN(yN) || Number.isNaN(dN)) {
   520→      return { status: "empty" as const };
   521→    }
   522→    if (src === "jalali") {
   523→      if (!isValidJalaaliDate(yN, shown.m, dN)) return { status: "invalid" as const };
   524→      return { status: "ok" as const, date: fromJalali(yN, shown.m, dN) };
   525→    }
   526→    if (src === "gregorian") {
   527→      if (yN < 1 || yN > 4000) return { status: "invalid" as const };
   528→      const dt = new Date(yN, shown.m - 1, dN);
   529→      if (
   530→        dt.getFullYear() !== yN ||
   531→        dt.getMonth() !== shown.m - 1 ||
   532→        dt.getDate() !== dN
   533→      ) {
   534→        return { status: "invalid" as const };
   535→      }
   536→      return { status: "ok" as const, date: dt };
   537→    }
   538→    // قمری
   539→    const dt = fromHijri(yN, shown.m, dN);
   540→    if (!dt) return { status: "invalid" as const };
   541→    const back = toHijri(dt);
   542→    if (back.hy !== yN || back.hm !== shown.m || back.hd !== dN) {
   543→      return { status: "invalid" as const };
   544→    }
   545→    return { status: "ok" as const, date: dt };
   546→  }, [src, shown.y, shown.m, shown.d]);
   547→
   548→  // دو خروجی دیگر
   549→  const outputs = useMemo(() => {
   550→    if (result.status !== "ok") return null;
   551→    const dt = result.date;
   552→    const wd = WEEKDAYS_FA[persianWeekday(dt)];
   553→    const list: { key: CalKey; title: string; line1: string; line2: string }[] = [];
   554→    if (src !== "jalali") {
   555→      const j = toJalali(dt);
   556→      list.push({
   557→        key: "jalali",
   558→        title: "تقویم شمسی",
   559→        line1: `${wd} ${faNum(j.jd)} ${JALALI_MONTHS[j.jm - 1]}`,
   560→        line2: `${faNum(j.jy)}`,
   561→      });
   562→    }
   563→    if (src !== "gregorian") {
   564→      list.push({
   565→        key: "gregorian",
   566→        title: "تقویم میلادی",
   567→        line1: `${wd} ${faNum(dt.getDate())} ${GREGORIAN_MONTHS[dt.getMonth()]}`,
   568→        line2: `${faNum(dt.getFullYear())}`,
   569→      });
   570→    }
   571→    if (src !== "hijri") {
   572→      const h = toHijri(dt);
   573→      list.push({
   574→        key: "hijri",
   575→        title: "تقویم قمری",
   576→        line1: `${wd} ${faNum(h.hd)} ${HIJRI_MONTHS[h.hm - 1]}`,
   577→        line2: `${faNum(h.hy)}`,
   578→      });
   579→    }
   580→    return list;
   581→  }, [result, src]);
   582→
   583→  const monthList = monthsOf(src);
   584→
   585→  return (
   586→    <div className="space-y-4">
   587→      {/* ورودی‌ها */}
   588→      <div className="min-w-0 space-y-4 rounded-3xl bg-card p-4 card-glow sm:p-6">
   589→        <div className="min-w-0">
   590→          <Label className="mb-2">تقویم مبدأ</Label>
   591→          <div className="flex min-w-0 flex-wrap gap-2">
   592→            {CONVERTER_CALS.map((c) => (
   593→              <button
   594→                key={c.key}
   595→                type="button"
   596→                onClick={() => switchSource(c.key)}
   597→                className={cn(
   598→                  "min-w-0 cursor-pointer rounded-full px-4 py-1.5 text-xs font-bold transition-all",
   599→                  src === c.key
   600→                    ? "bg-gradient-to-l from-violet-500 to-purple-500 text-white shadow-md shadow-violet-500/25"
   601→                    : "bg-secondary text-secondary-foreground hover:bg-accent"
   602→                )}
   603→              >
   604→                {c.label}
   605→              </button>
   606→            ))}
   607→          </div>
   608→        </div>
   609→
   610→        <div className="grid min-w-0 grid-cols-3 gap-2 sm:gap-3">
   611→          <div className="min-w-0 space-y-1.5">
   612→            <Label htmlFor="conv-year">سال</Label>
   613→            <Input
   614→              id="conv-year"
   615→              inputMode="numeric"
   616→              value={shown.y}
   617→              onChange={(e) => update({ y: digitsOnly(e.target.value) })}
   618→              placeholder="مثلاً ۱۴۰۴"
   619→              className="rounded-xl"
   620→            />
   621→          </div>
   622→          <div className="min-w-0 space-y-1.5">
   623→            <Label>ماه</Label>
   624→            <Select
   625→              value={shown.m ? String(shown.m) : ""}
   626→              onValueChange={(v) => update({ m: Number(v) })}
   627→              dir="rtl"
   628→            >
   629→              <SelectTrigger className="w-full rounded-xl">
   630→                <SelectValue placeholder="ماه" />
   631→              </SelectTrigger>
   632→              <SelectContent dir="rtl">
   633→                {monthList.map((name, i) => (
   634→                  <SelectItem key={i + 1} value={String(i + 1)}>
   635→                    {name}
   636→                  </SelectItem>
   637→                ))}
   638→              </SelectContent>
   639→            </Select>
   640→          </div>
   641→          <div className="min-w-0 space-y-1.5">
   642→            <Label htmlFor="conv-day">روز</Label>
   643→            <Input
   644→              id="conv-day"
   645→              inputMode="numeric"
   646→              value={shown.d}
   647→              onChange={(e) => update({ d: digitsOnly(e.target.value) })}
   648→              placeholder="مثلاً ۱۵"
   649→              className="rounded-xl"
   650→            />
   651→          </div>
   652→        </div>
   653→
   654→        <div className="flex min-w-0 flex-wrap items-center gap-2">
   655→          <Button
   656→            type="button"
   657→            variant="secondary"
   658→            size="sm"
   659→            onClick={resetToToday}
   660→            className="gap-1.5 rounded-full"
   661→          >
   662→            <RotateCcw className="h-3.5 w-3.5" />
   663→            امروز
   664→          </Button>
   665→          {result.status === "invalid" && (
   666→            <Chip className="shrink-0 border border-red-500/30 bg-red-500/10 text-red-500">
   667→              تاریخ نامعتبر است
   668→            </Chip>
   669→          )}
   670→          {result.status === "empty" && (
   671→            <span className="text-xs text-muted-foreground">
   672→              سال، ماه و روز را وارد کن تا تبدیل شود
   673→            </span>
   674→          )}
   675→        </div>
   676→      </div>
   677→
   678→      {/* خروجی دو تقویم دیگر */}
   679→      {outputs && outputs.length > 0 && (
   680→        <div className="grid min-w-0 gap-3 sm:grid-cols-2">
   681→          {outputs.map((o) => (
   682→            <motion.div
   683→              key={`${src}-${o.key}`}
   684→              initial={{ opacity: 0, y: 8 }}
   685→              animate={{ opacity: 1, y: 0 }}
   686→              className="min-w-0 rounded-3xl bg-card p-4 card-glow sm:p-5"
   687→            >
   688→              <div className="flex items-center gap-2 text-violet-500">
   689→                <ArrowLeftRight className="h-4 w-4" />
   690→                <span className="text-xs font-extrabold">{o.title}</span>
   691→              </div>
   692→              <p className="mt-2 text-base font-black text-foreground sm:text-lg">
   693→                {o.line1}
   694→              </p>
   695→              <p className="text-sm font-bold text-muted-foreground">{o.line2}</p>
   696→            </motion.div>
   697→          ))}
   698→        </div>
   699→      )}
   700→    </div>
   701→  );
   702→}
